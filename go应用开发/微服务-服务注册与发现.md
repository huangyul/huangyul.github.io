## etcd 

### 服务注册与发现

基本用法：

```go
// 注册服务
leaseResp, err := cli.Grant(context.Background(), 5)
cli.Put(context.Background(), key, addr, clientv3.WithLease(leaseResp.ID))
// 自动续约
keepalive, err := cli.keepAlive(context.Backgroud(), leaseResp.ID)

// 发现服务
etcdRes, err := cli.Get(context.Background(), key)
for _, kv := range etcdRes.Kvs {
	fmt.Printf("key: %s, value: %s \r\n", string(kv.Key), string(kv.Value))
}
```

使用`grpc.resolver`接口:

```go
  // 注册服务
  em, err := endpoints.NewManager(ec, ServerName)
  if err != nil {
    panic(err)
  }
  leaseResp, err := ec.Grant(context.Background(), 5)
  if err != nil {
    panic(err)
  }
  err = em.AddEndpoint(context.Background(), ServerName+"/"+addr, endpoints.Endpoint{
    Addr: addr,
  }, clientv3.WithLease(leaseResp.ID))
  if err != nil {
    panic(err)
  }
  // 自动续约
  kaResp, err := ec.KeepAlive(context.Background(), leaseResp.ID)
  if err != nil {
    panic(err)
  }
  go func() {
    for r := range kaResp {
      fmt.Println(r)
    }
  }()


// 发现服务
  reso, err := resolver.NewBuilder(ec)
  if err != nil {
    panic(err)
  }
  conn, err := grpc.NewClient("etcd:///service/hello", grpc.WithResolvers(reso), grpc.WithTransportCredentials(insecure.NewCredentials()))
```
